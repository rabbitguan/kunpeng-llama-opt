cmake_minimum_required(VERSION 3.14)
project(tmac_baseline_harness LANGUAGES C CXX)

if(NOT LLAMA_SOURCE)
    message(FATAL_ERROR "Set LLAMA_SOURCE to the pinned llama.cpp submodule")
endif()

set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
if(TMAC_CPU_BASELINE)
    set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
else()
    set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory(${LLAMA_SOURCE} llama)

if(TMAC_CPU_BASELINE)
    add_executable(llama-cpu-smoke cpu-baseline.cpp)
    target_link_libraries(llama-cpu-smoke PRIVATE llama ggml-cpu)
    target_compile_features(llama-cpu-smoke PRIVATE cxx_std_17)
else()
    add_executable(tmac-operator operator.cpp)
    target_link_libraries(tmac-operator PRIVATE ggml)
    target_include_directories(tmac-operator PRIVATE ${LLAMA_SOURCE}/ggml/include)
    target_compile_features(tmac-operator PRIVATE cxx_std_17)

    add_executable(tmac-logits logits.cpp)
    target_link_libraries(tmac-logits PRIVATE llama)
    target_compile_features(tmac-logits PRIVATE cxx_std_17)
endif()

if(TMAC_CACHE_SCHEDULE)
    add_executable(tmac-cache-operator operator.cpp)
    target_link_libraries(tmac-cache-operator PRIVATE ggml)
    target_include_directories(tmac-cache-operator PRIVATE ${LLAMA_SOURCE}/ggml/include)
    target_compile_features(tmac-cache-operator PRIVATE cxx_std_17)
    target_compile_definitions(tmac-cache-operator PRIVATE TMAC_CACHE_SCHEDULE=1)
    target_compile_options(tmac-cache-operator PRIVATE -march=armv8.2-a+fp16)
endif()
