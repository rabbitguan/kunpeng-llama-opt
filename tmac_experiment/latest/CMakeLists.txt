cmake_minimum_required(VERSION 3.22)
project(tmac_latest_checks LANGUAGES C CXX)
if(NOT LLAMA_SOURCE)
    message(FATAL_ERROR "Set LLAMA_SOURCE to the prepared c622f62 source tree")
endif()
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_SOURCE} llama)
if(TMAC_FUSED)
    if(NOT GGML_TMAC)
        message(FATAL_ERROR "TMAC_FUSED requires GGML_TMAC")
    endif()
    target_compile_definitions(ggml-cpu PRIVATE GGML_TMAC_FUSED)
endif()
if(TMAC_DIAGNOSTIC_BACKEND_SOURCE)
    if(NOT GGML_TMAC)
        message(FATAL_ERROR "The diagnostic source override requires GGML_TMAC")
    endif()
    get_target_property(cpu_sources ggml-cpu SOURCES)
    set(original_sources ${cpu_sources})
    list(FILTER cpu_sources EXCLUDE REGEX "(^|/)tmac/tmac\\.cpp$")
    list(LENGTH original_sources count_before)
    list(LENGTH cpu_sources count_after)
    math(EXPR expected_count "${count_after} + 1")
    if(NOT count_before EQUAL expected_count)
        message(FATAL_ERROR "Cannot identify the original T-MAC backend source")
    endif()
    set_property(TARGET ggml-cpu PROPERTY SOURCES ${cpu_sources})
    target_sources(ggml-cpu PRIVATE ${TMAC_DIAGNOSTIC_BACKEND_SOURCE})
    target_include_directories(ggml-cpu PRIVATE ${LLAMA_SOURCE}/ggml/src/ggml-cpu/tmac)
endif()
add_executable(tmac-latest-check check.cpp)
target_link_libraries(tmac-latest-check PRIVATE ggml ggml-cpu)
target_compile_features(tmac-latest-check PRIVATE cxx_std_17)
add_executable(tmac-latest-logits logits.cpp)
target_link_libraries(tmac-latest-logits PRIVATE llama)
target_compile_features(tmac-latest-logits PRIVATE cxx_std_17)
if(GGML_TMAC)
    get_target_property(tmac_sources ggml-cpu SOURCES)
    list(FILTER tmac_sources INCLUDE REGEX "(^|/)kernels\\.cc$")
    list(GET tmac_sources 0 tmac_reference)
    add_executable(tmac-fused-check fused-check.cpp)
    target_link_libraries(tmac-fused-check PRIVATE t_mac_no_tvm)
    target_compile_features(tmac-fused-check PRIVATE cxx_std_17)
    target_compile_options(tmac-fused-check PRIVATE $<TARGET_PROPERTY:ggml-cpu,COMPILE_OPTIONS>)
    target_compile_definitions(tmac-fused-check PRIVATE TMAC_REFERENCE_SOURCE="${tmac_reference}")
endif()
